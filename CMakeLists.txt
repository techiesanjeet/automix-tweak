cmake_minimum_required(VERSION 3.16)
project(TaurusSmartMixCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(smartmix SmartMixCore/src/SmartMix.cpp SmartMixCore/src/SmartMixBridge.cpp SmartMixCore/src/PreparedPCM.cpp SmartMixCore/src/Resampler.cpp SmartMixCore/src/AudioInputAdapter.cpp SmartMixCore/src/BeatThisSpectrogram.cpp SmartMixCore/src/BeatThisPostprocessor.cpp SmartMixCore/src/BeatThisChunking.cpp)
target_include_directories(smartmix PUBLIC SmartMixCore/include)
target_compile_options(smartmix PRIVATE -Wall -Wextra -Wpedantic)
add_executable(smartmix_tests SmartMixCore/tests/SmartMixTests.cpp)
target_link_libraries(smartmix_tests PRIVATE smartmix)
enable_testing()
add_test(NAME smartmix_tests COMMAND smartmix_tests)
